if(NOT WIN32)
    message(FATAL_ERROR "The desktop launcher currently targets Windows")
endif()
set(WOH_JNI_INCLUDE "" CACHE PATH "Directory containing jni.h")
if(NOT EXISTS "${WOH_JNI_INCLUDE}/jni.h")
    message(FATAL_ERROR "Set WOH_JNI_INCLUDE to the JDK or Android NDK JNI include directory")
endif()
set(WOH_FFMPEG_ROOT "" CACHE PATH "Pinned LGPL FFmpeg SDK directory")
if(NOT EXISTS "${WOH_FFMPEG_ROOT}/include/libavformat/avformat.h")
    message(FATAL_ERROR "Set WOH_FFMPEG_ROOT to the LGPL FFmpeg SDK directory")
endif()
add_library(woh_desktop SHARED ../../android/app/src/main/cpp/bridge.cpp assets.cpp video.cpp)
target_include_directories(woh_desktop PRIVATE "${WOH_JNI_INCLUDE}" "${WOH_JNI_INCLUDE}/win32")
target_link_libraries(woh_desktop PRIVATE woh_runtime)
target_include_directories(woh_desktop PRIVATE "${WOH_FFMPEG_ROOT}/include")
foreach(component avformat avcodec avutil swscale)
    target_link_libraries(woh_desktop PRIVATE "${WOH_FFMPEG_ROOT}/lib/lib${component}.dll.a")
endforeach()
set_target_properties(woh_desktop PROPERTIES OUTPUT_NAME woh_android PREFIX "" WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_link_options(woh_desktop PRIVATE -static)
endif()
add_executable(mahoyo_windows WIN32 launcher.cpp)
add_executable(mahoyo_sync WIN32 launcher.cpp)
target_compile_definitions(mahoyo_sync PRIVATE WOH_SAVE_SYNC)
target_link_libraries(mahoyo_sync PRIVATE shell32 user32)
set_target_properties(mahoyo_sync PROPERTIES OUTPUT_NAME SaveSync)
target_link_libraries(mahoyo_windows PRIVATE shell32 user32)
set_target_properties(mahoyo_windows PROPERTIES OUTPUT_NAME Mahoyo)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_link_options(mahoyo_windows PRIVATE -municode -static)
    target_link_options(mahoyo_sync PRIVATE -municode -static)
endif()
