cmake_minimum_required(VERSION 3.22.1)
project(woh_android LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(WOH_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
add_subdirectory("${WOH_ROOT}/runtime" "${CMAKE_CURRENT_BINARY_DIR}/runtime")
add_library(woh_android SHARED bridge.cpp asset_bridge.cpp steam_chunk_bridge.cpp "${WOH_ROOT}/dependencies/lzma/LzmaDec.c")
if(ANDROID_ABI STREQUAL "arm64-v8a")
  target_sources(woh_android PRIVATE "${WOH_ROOT}/dependencies/lzma/LzmaDecOpt.S")
  set_source_files_properties("${WOH_ROOT}/dependencies/lzma/LzmaDec.c" PROPERTIES COMPILE_DEFINITIONS Z7_LZMA_DEC_OPT)
endif()
target_include_directories(woh_android PRIVATE "${WOH_ROOT}/runtime" "${WOH_ROOT}/dependencies/lzma")
target_link_libraries(woh_android PRIVATE woh_runtime log jnigraphics dl m)
