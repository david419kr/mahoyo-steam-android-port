cmake_minimum_required(VERSION 3.22.1)
project(woh_android LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(WOH_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
add_subdirectory("${WOH_ROOT}/runtime" "${CMAKE_CURRENT_BINARY_DIR}/runtime")
set(WOH_AVIF_DEPS "${WOH_ROOT}/.deps/android/arm64-v8a" CACHE PATH "Staged software AVIF decoder")
file(READ "${WOH_AVIF_DEPS}/lock.json" WOH_AVIF_LOCK)
foreach(codec avif dav1d yuv)
  if(NOT EXISTS "${WOH_AVIF_DEPS}/lib/lib${codec}.a")
    message(FATAL_ERROR "Run scripts/prepare-avif-deps.py --packages <vcpkg/packages> before building")
  endif()
  file(SHA256 "${WOH_AVIF_DEPS}/lib/lib${codec}.a" WOH_ACTUAL)
  string(JSON WOH_EXPECTED GET "${WOH_AVIF_LOCK}" stagedLibraries "${codec}")
  if(NOT WOH_ACTUAL STREQUAL WOH_EXPECTED)
    message(FATAL_ERROR "Staged AVIF ${codec} differs from dependency lock")
  endif()
  add_library(woh_${codec} STATIC IMPORTED)
  set_target_properties(woh_${codec} PROPERTIES IMPORTED_LOCATION "${WOH_AVIF_DEPS}/lib/lib${codec}.a")
endforeach()
string(JSON WOH_HEADER_COUNT LENGTH "${WOH_AVIF_LOCK}" stagedHeaders)
math(EXPR WOH_HEADER_LAST "${WOH_HEADER_COUNT}-1")
foreach(index RANGE 0 ${WOH_HEADER_LAST})
  string(JSON header MEMBER "${WOH_AVIF_LOCK}" stagedHeaders ${index})
  string(JSON WOH_EXPECTED GET "${WOH_AVIF_LOCK}" stagedHeaders "${header}")
  file(SHA256 "${WOH_AVIF_DEPS}/include/${header}" WOH_ACTUAL)
  if(NOT WOH_ACTUAL STREQUAL WOH_EXPECTED)
    message(FATAL_ERROR "Staged AVIF header ${header} differs from dependency lock")
  endif()
endforeach()
add_library(woh_android SHARED bridge.cpp avif_bridge.cpp)
target_include_directories(woh_android PRIVATE "${WOH_ROOT}/runtime" "${WOH_AVIF_DEPS}/include")
target_link_libraries(woh_android PRIVATE woh_runtime woh_avif woh_dav1d woh_yuv log jnigraphics dl m)
